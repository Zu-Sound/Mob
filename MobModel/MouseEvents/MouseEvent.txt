        List<int> selectedIndices = new List<int>();
        private int anchorIndex = -1;
        private bool wasDraggingToExplorer = false;
        private bool dragStarted = false;
        private bool dragSuppressSelection = false;
        private bool lastActionWasShift = false;

        private void ListBox1_MouseDown(object sender, MouseEventArgs e)
        {
            if (e.Button != MouseButtons.Left)
            {
                mouseDownPoint = Point.Empty;
                return;
            }

            dragStarted = false;
            wasDraggingToExplorer = false;

            if (this.Owner is not Form1 fm)
                return;

            fm.dropSource = 2;

            // -------------------------
            // �� �N���b�N�ʒu�̎擾�i�n�C�u���b�h�����j
            // -------------------------
            int Qind;

            if (lastActionWasShift)
            {
                Qind = ListBox1.IndexFromPoint(e.Location);
                lastActionWasShift = false;
            }
            else
            {
                Qind = GetClickedIndex(ListBox1, e);
            }

            if (Qind >= 0)
                mouseDownPoint = new Point(e.X, e.Y);

            // -------------------------
            // �󔒃N���b�N
            // -------------------------
            if (Qind < 0)
            {
                selectedIndices.Clear();
                foreach (int idx in ListBox1.SelectedIndices)
                    selectedIndices.Add(idx);

                anchorIndex = selectedIndices.Count > 0
                    ? selectedIndices[selectedIndices.Count - 1]
                    : -1;

                return;
            }

            bool shiftHeld = (ModifierKeys & Keys.Shift) == Keys.Shift;
            bool ctrlHeld = (ModifierKeys & Keys.Control) == Keys.Control;

            // -------------------------
            // Ctrl �g�O��
            // -------------------------
            if (ctrlHeld)
            {
                bool isSelected = selectedIndices.Contains(Qind);

                if (isSelected)
                {
                    ListBox1.SetSelected(Qind, false);
                    selectedIndices.Remove(Qind);
                }
                else
                {
                    ListBox1.SetSelected(Qind, true);
                    selectedIndices.Add(Qind);
                }

                return;
            }

            // -------------------------
            // Shift �͈͑I���iMouseDown �Ŋ����j
            // -------------------------
            if (shiftHeld)
            {
                // �A���J�[���ݒ�Ȃ猻�݈ʒu
                if (anchorIndex < 0)
                    anchorIndex = Qind;

                // �� ����͈̔́i�k��OK�j
                int start = Math.Min(anchorIndex, Qind);
                int end = Math.Max(anchorIndex, Qind);

                int savedTopIndex = ListBox1.TopIndex;

                ListBox1.BeginUpdate();
                ListBox1.SelectedIndices.Clear();
                selectedIndices.Clear();

                for (int i = start; i <= end; i++)
                {
                    ListBox1.SetSelected(i, true);
                    selectedIndices.Add(i);
                }

                ListBox1.EndUpdate();
                ListBox1.TopIndex = savedTopIndex;

                selectionCurrent = Qind;
                lastActionWasShift = true;
                return;
            }

            // -------------------------
            // �����I�𒆂̃N���b�N
            // -------------------------
            if (selectedIndices.Count > 1)
            {
                if (selectedIndices.Contains(Qind))
                {
                    // �� �͈͓��N���b�N �� �����I���ێ��iTopIndex �ۑ������j
                    int savedTopIndex = ListBox1.TopIndex;

                    ListBox1.BeginUpdate();
                    ListBox1.SelectedIndices.Clear();
                    foreach (int i in selectedIndices)
                        ListBox1.SetSelected(i, true);
                    ListBox1.EndUpdate();

                    ListBox1.TopIndex = savedTopIndex;

                    selectionCurrent = Qind;
                    return;
                }
                else
                {
                    // �� �͈͊O�N���b�N �� �P�ƑI���ɖ߂��i������Ԃ̂݁j
                    selectedIndices.Clear();
                    selectedIndices.Add(Qind);

                    anchorIndex = Qind;
                    selectionCurrent = Qind;

                    return;
                }
            }

            // -------------------------
            // �� �P�ƑI��
            // -------------------------
            anchorIndex = Qind;

            selectedIndices.Clear();
            selectedIndices.Add(Qind);

            selectionCurrent = Qind;
        }

        private int GetClickedIndex(ListBox lbx, MouseEventArgs e)
        {
            int itemHeight = lbx.ItemHeight;
            int top = lbx.TopIndex;

            int clickedRow = e.Y / itemHeight;
            int index = top + clickedRow;

            if (index < 0 || index >= lbx.Items.Count)
                return -1;

            return index;
        }


        private void ListBox1_MouseMove(object sender, MouseEventArgs e)
        {
            if (this.Owner is not Form1 fm)
                return;

            string MPath = Path.Combine(fm.appBasePath, "music");

            if (mouseDownPoint != Point.Empty)
            {
                Rectangle moveRect = new Rectangle(
                    mouseDownPoint.X - SystemInformation.DragSize.Width / 3,
                    mouseDownPoint.Y - SystemInformation.DragSize.Height / 3,
                    SystemInformation.DragSize.Width,
                    SystemInformation.DragSize.Height);

                if (!moveRect.Contains(e.X, e.Y))
                {
                    ListBox lbx = (ListBox)sender;
                    int itemIndex = lbx.IndexFromPoint(mouseDownPoint);
                    if (itemIndex < 0) return;

                    Point pp = Control.MousePosition;
                    pp = ListBox1.PointToClient(pp);
                    int Qind = ListBox1.IndexFromPoint(pp);
                    if (Qind != -1)
                    {
                        dragStarted = true;
                        dragSuppressSelection = true;

                        // �t�@�C���p�X�z��iExplorer�p�j
                        draggedFilePaths = selectedIndices
                            .Select(i => ListBox1.Items[i]?.ToString())
                            .Where(name => !string.IsNullOrEmpty(name))
                            .Select(name => Path.Combine(MPath, name!))
                            .ToArray();

                        // �t�@�C�����z��i�t�H�[���ԗp�j: ��null��string�����𒊏o
                        string[] names = selectedIndices
                            .Select(i => ListBox1.Items[i])   // object �����o��
                            .OfType<string>()                 // string �ɍi�荞�݁inull/���^�͏��O�j
                            .Where(s => !string.IsNullOrWhiteSpace(s))
                            .ToArray();

                        if (draggedFilePaths.Length > 0 || names.Length > 0)
                        {
                            wasDraggingToExplorer = true;

                            var data = new DataObject();
                            if (draggedFilePaths.Length > 0)
                                data.SetData(DataFormats.FileDrop, draggedFilePaths);
                            if (names.Length > 0)
                                data.SetData(DataFormats.Text, string.Join(Environment.NewLine, names));

                            data.SetData("Mob.Ddn", fm.dropSource);

                            // �����p�ɃR�s�[���t���p�X���n��
                            if (draggedFilePaths.Length > 0)
                                data.SetData("Mob.SourcePaths", draggedFilePaths);

                            // ��� Copy �Œ�
                            DragDropEffects result = DoDragDrop(data, DragDropEffects.Copy);

                            if (result == DragDropEffects.None)
                            {
                                draggedFilePaths = null;
                                selectedIndices.Clear();
                                dragStarted = false;
                                wasDraggingToExplorer = false;
                            }
                        }
                    }

                    mouseDownPoint = Point.Empty;
                }
            }
        }

        private void ListBox1_MouseUp(object sender, MouseEventArgs e)
        {
            if (this.Owner is not Form1 fm)
                return;

            bool shiftHeld = (ModifierKeys & Keys.Shift) == Keys.Shift;
            bool ctrlHeld = (ModifierKeys & Keys.Control) == Keys.Control;

            // -------------------------
            // �ʏ�N���b�N���� UI ���m��
            // -------------------------
            if (e.Button == MouseButtons.Left &&
                !dragStarted &&
                !shiftHeld &&
                !ctrlHeld)
            {
                int Qind = GetClickedIndex(ListBox1, e);

                if (Qind >= 0)
                {
                    ListBox1.SelectedIndices.Clear();
                    ListBox1.SetSelected(Qind, true);

                    selectedIndices.Clear();
                    selectedIndices.Add(Qind);

                    anchorIndex = Qind;
                    selectionCurrent = Qind;
                }
            }
            else
            {
                // Shift/Ctrl �̂Ƃ��� UI ��G��Ȃ��iMouseDown �Ŋm��ς݁j
                if (selectedIndices.Count == 1)
                {
                    anchorIndex = selectedIndices[0];
                    selectionCurrent = selectedIndices[0];
                }
                else if (selectedIndices.Count > 1)
                {
                    selectionCurrent = selectedIndices[selectedIndices.Count - 1];
                }
                else
                {
                    anchorIndex = -1;
                    selectionCurrent = -1;
                }
            }

            Label_All.Text = $"{ListBox1.SelectedIndex + 1} / {ListBox1.Items.Count}";
            fm.dropSource = 10;

            dragStarted = false;
            dragSuppressSelection = false;
            mouseDownPoint = Point.Empty;
        }
